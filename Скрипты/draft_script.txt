def getMailRecipients(String groupName) {
    def mailCfg = readJSON file: 'mailing-list.json', returnPojo: true

    if (!mailCfg.containsKey(groupName)) {
        error "Группа '${groupName}' не найдена в mailing-list.json"
    }

    return mailCfg[groupName].join(',')
}

def sendEmailNotification(String title, String bodyHtml, boolean attachLog) {
    def cfg = BASE_CONFIG[params.BASE_NAME]
    def groupName = cfg?.mailGroup ?: 'draft'

    def recipients = getMailRecipients(groupName)

    emailext(
        subject: "${title} ${params.BASE_NAME}",
        body: bodyHtml,
        mimeType: 'text/html',
        from: 'jenkins@mechel.com',
        to: recipients,
        attachLog: attachLog
    )
}
